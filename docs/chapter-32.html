def get_connection() -> sqlite3.Connection:
    config.ensure_data_dir()
    conn = sqlite3.connect(config.DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn

def init_db() -> None:
    conn = get_connection()
    try:
        with conn:
            conn.executescript(SCHEMA)
    finally:
        conn.close()

32.7.2 关键代码（来源回查）

def _best_paragraph(content: str, keywords: list[str]) -> str:
    paragraphs = [p.strip() for p in re.split(r"\n\s*\n", content) if p.strip()]
    best, best_score = paragraphs[0], -1
    for p in paragraphs:
        score = sum(1 for k in keywords if k.lower() in p.lower())
        if score > best_score:
            best, best_score = p, score
    return best[:400] if best_score else paragraphs[0][:200]

只保留回答中实际标注的引用（兜底：一个都没标则全保留）：

cited = {int(m) for m in re.findall(r"\[(\d+)\]", answer) if int(m) <= len(sources)}
