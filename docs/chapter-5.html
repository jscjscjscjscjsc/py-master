01. 字符串索引与切片 ▼

索引（从 0 开始）

s = "Hello Python"
print(s[0])   # H（第一个字符）
print(s[6])   # P（第七个字符，空格也算一个字符）
print(s[-1])  # n（最后一个字符）
print(s[-3])  # h（倒数第三个）

切片 [start:stop:step]

s = "Hello Python"
# 三个参数都可以省略
print(s[0:5])    # Hello（索引 0~4，含头不含尾）
print(s[6:])     # Python（从索引6到末尾）
print(s[:5])     # Hello（从开头到索引4）
print(s[::2])    # HloPto（步长为2，隔一个取一个）
print(s[::-1])   # nohtyP olleH（反转字符串！经典技巧）

切片的三个参数

# s[start:stop:step]
# start: 起始索引（默认0）
# stop:  结束索引（默认末尾），不包含该索引位置
# step:  步长（默认1），负数表示反向

s = "0123456789"
print(s[2:8])     # "234567"
print(s[2:8:2])   # "246"
print(s[8:2:-1])  # "876543"（反向切片）

切片 vs 索引：越界行为不同

s = "Hello"
# 索引越界 → 报错
# print(s[100])  # IndexError!

# 切片越界 → 不报错，返回空或截断到末尾
print(s[0:100])  # "Hello"（安全！）
print(s[100:])   # ""（空字符串）

字符串不可变性（复习）

s = "Hello"
# s[0] = "h"  # ❌ 报错
s = "h" + s[1:]  # ✅ 创建新字符串

02. 字符串常用方法（上） ▼

大小写转换

s = "Hello World"
print(s.upper())       # HELLO WORLD
print(s.lower())       # hello world
print(s.title())       # Hello World（每个单词首字母大写）
print(s.capitalize())  # Hello world（只有第一个字符大写）
print(s.swapcase())    # hELLO wORLD（大小写互换）

# 实用场景
user_input = "  YES  "
if user_input.strip().lower() == "yes":  # 不区分大小写
    print("确认")

去除空白

s = "  hello world  \n"
print(repr(s.strip()))   # 'hello world'（去两端）
print(repr(s.lstrip()))  # 'hello world  \n'（去左侧）
print(repr(s.rstrip()))  # '  hello world'（去右侧）

查找与替换

s = "Hello World, Hello Python"
# find vs index
print(s.find("World"))     # 6（找到返回索引）
print(s.find("Java"))      # -1（没找到返回-1，不报错）
# print(s.index("Java"))   # ❌ ValueError!（没找到会报错）

# rfind: 从右侧查找
print(s.rfind("Hello"))    # 13（最后一次出现的位置）

# count: 统计次数
print(s.count("Hello"))    # 2
print(s.count("l"))        # 5

# replace
print(s.replace("Hello", "你好"))  # 你好 World, 你好 Python
print(s.replace("Hello", "你好", 1))  # 你好 World, Hello Python（只替换第一处）

03. 字符串常用方法（下） ▼

字符串判断方法（返回 True/False）

# 内容类型判断
print("123".isdigit())    # True（全数字）
print("abc".isalpha())    # True（全字母）
print("abc123".isalnum()) # True（字母或数字）
print("  ".isspace())     # True（全空白字符）
print("Hello".isascii())  # True（全 ASCII 字符）

# 大小写判断
print("hello".islower())  # True
print("HELLO".isupper())  # True
print("Hello".istitle())  # True

# 前后缀判断
print("hello.py".startswith("hello"))  # True
print("hello.py".endswith(".py"))      # True（判断文件类型常用）

分割与连接

# split: 字符串 → 列表（重要！）
s = "苹果,香蕉,橙子,葡萄"
fruits = s.split(",")
print(fruits)  # ['苹果', '香蕉', '橙子', '葡萄']

# 限制分割次数
print(s.split(",", 2))  # ['苹果', '香蕉', '橙子,葡萄']

# rsplit: 从右侧分割
print(s.rsplit(",", 1))  # ['苹果,香蕉,橙子', '葡萄']

# splitlines: 按行分割
text = "第一行\n第二行\n第三行"
print(text.splitlines())  # ['第一行', '第二行', '第三行']

# join: 列表 → 字符串（split 的逆操作）
print(" | ".join(fruits))   # 苹果 | 香蕉 | 橙子 | 葡萄
print("".join(["H","e","l","l","o"]))  # Hello

partition: 分割成三部分

email = "user@example.com"
local, at, domain = email.partition("@")
print(local)   # user
print(at)      # @
print(domain)  # example.com

04. 字符串格式化 ▼

四种格式化方式（推荐 f-string）

name = "Alice"
age = 25

# 1. f-string（Python 3.6+，⭐最推荐！）
print(f"我叫{name}，明年{age + 1}岁")

# 2. str.format() 方法
print("我叫{}，今年{}岁".format(name, age))
print("我叫{0}，今年{1}岁".format(name, age))  # 按位置
print("我叫{n}，今年{a}岁".format(n=name, a=age))  # 按名称

# 3. % 格式化（旧式，不推荐新代码使用）
print("我叫%s，今年%d岁" % (name, age))

# 4. 字符串拼接（最差，不用）
print("我叫" + name + "，今年" + str(age) + "岁")

f-string 高级格式化

price = 12345.6789
# 小数控制
print(f"{price:.2f}")    # 12345.68（保留2位小数，四舍五入）
print(f"{price:.0f}")    # 12346

# 千分位
print(f"{price:,.2f}")   # 12,345.68

# 对齐与填充
name = "Bob"
print(f"|{name:<10}|")   # |Bob       |（左对齐，占10格）
print(f"|{name:>10}|")   # |       Bob|（右对齐）
print(f"|{name:^10}|")   # |   Bob    |（居中对齐）
print(f"|{name:*^10}|")  # |***Bob****|（*填充居中）

# 进制转换
n = 255
print(f"{n:b}")   # 11111111（二进制）
print(f"{n:x}")   # ff（十六进制）
print(f"{n:#x}")  # 0xff

# 百分比
rate = 0.857
print(f"{rate:.1%}")  # 85.7%

format() 高级用法

# 按位置
print("{1} 比 {0} 大".format(3, 5))  # 5 比 3 大

# 按名称
print("坐标: ({x}, {y})".format(x=10, y=20))

# 解包字典
data = {"name": "Tom", "score": 95}
print("{name}: {score}分".format(**data))
