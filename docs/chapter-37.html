02. 2.1 Day 06｜异步编程与多供应商协议 ▼

2.1.1 当日目标

学生先理解“等待网络时让出执行权”，再写并发。验收不是只看速度，而是能解释事件循环、协程、任务、并发限制和异常隔离。

2.1.2 用等待任务体感并发

python
import asyncio
from time import perf_counter


async def simulated_request(name: str, seconds: float) -> str:
    await asyncio.sleep(seconds)
    return f"{name} 完成"


async def main() -> None:
    started = perf_counter()
    results = await asyncio.gather(
        simulated_request("A", 1.0),
        simulated_request("B", 1.5),
        simulated_request("C", 0.5),
    )
    print(results)
    print(f"总耗时: {perf_counter() - started:.2f}s")


if __name__ == "__main__":
    asyncio.run(main())

逐层解释：

  1. 调用 async def 得到协程对象，await 才真正等待其完成。
  2. asyncio.sleep 模拟网络等待，它不会占住事件循环。
  3. gather 并发调度三个协程，总耗时接近最慢任务，而不是三者之和。
  4. 异步适合大量 I/O 等待，不会自动加速 CPU 密集计算。

2.1.3 统一配置与响应对象

python
from dataclasses import dataclass
import os


@dataclass(frozen=True)
class ProviderConfig:
    name: str
    base_url: str
    api_key: str
    model: str


@dataclass(frozen=True)
class ChatResult:
    text: str
    model: str
    provider: str
    prompt_tokens: int
    completion_tokens: int


def load_provider(prefix: str) -> ProviderConfig:
    values = {
        "base_url": os.getenv(f"{prefix}_BASE_URL", ""),
        "api_key": os.getenv(f"{prefix}_API_KEY", ""),
        "model": os.getenv(f"{prefix}_CHAT_MODEL", ""),
    }
    missing = [key for key, value in values.items() if not value]
    if missing:
        raise RuntimeError(f"{prefix} 缺少配置: {', '.join(missing)}")
    return ProviderConfig(prefix.lower(), **values)

统一对象让上层不依赖某家供应商的环境变量名和原始 JSON 结构。

2.1.4 异步 LLM 客户端

python
import httpx


class AsyncLLMClient:
    def __init__(self, config: ProviderConfig):
        self.config = config
        self.client = httpx.AsyncClient(
            base_url=config.base_url.rstrip("/"),
            headers={"Authorization": f"Bearer {config.api_key}"},
            timeout=httpx.Timeout(60.0, connect=10.0),
            limits=httpx.Limits(max_connections=20, max_keepalive_connections=10),
        )

    async def chat(self, messages: list[dict[str, str]]) -> ChatResult:
        response = await self.client.post(
            "/chat/completions",
            json={"model": self.config.model, "messages": messages, "temperature": 0.2},
        )
        response.raise_for_status()
        data = response.json()
        usage = data.get("usage") or {}
        return ChatResult(
            text=data["choices"][0]["message"]["content"],
            model=data.get("model", self.config.model),
            provider=self.config.name,
            prompt_tokens=int(usage.get("prompt_tokens", 0)),
            completion_tokens=int(usage.get("completion_tokens", 0)),
        )

    async def close(self) -> None:
        await self.client.aclose()

关键解释：

  • 客户端在应用生命周期内复用，才能利用连接池；不要在热点循环里反复创建。
  • raise_for_status() 将非成功 HTTP 状态变成明确异常。
  • 响应归一化后，上层只读取 ChatResult。
  • 日志不得输出 Authorization。

2.1.5 有上限的多供应商并发

python
async def compare_providers(clients: list[AsyncLLMClient], question: str, limit: int = 3) -> list[dict]:
    semaphore = asyncio.Semaphore(limit)

    async def ask_one(client: AsyncLLMClient) -> dict:
        async with semaphore:
            started = perf_counter()
            try:
                result = await client.chat([{"role": "user", "content": question}])
                return {"ok": True, "provider": result.provider, "text": result.text, "latency": perf_counter() - started}
            except Exception as exc:
                return {"ok": False, "provider": client.config.name, "error": type(exc).__name__, "latency": perf_counter() - started}

    return await asyncio.gather(*(ask_one(client) for client in clients))

Semaphore 保护本机连接池和上游额度。每个任务捕获自己的异常，避免一家失败导致整批结果丢失。

2.2.2 应用生命周期与依赖

python
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request


@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.llm = AsyncLLMClient(load_provider("LLM"))
    yield
    await app.state.llm.close()


app = FastAPI(title="AI Chat Backend", lifespan=lifespan)


def get_llm(request: Request) -> AsyncLLMClient:
    return request.app.state.llm

应用启动时创建一次客户端，关闭时释放连接。这个模式比模块导入时直接创建网络客户端更容易测试。

2.2.3 /health 和 /chat

python
from fastapi import Depends, HTTPException
from pydantic import BaseModel, Field


class Message(BaseModel):
    role: str
    content: str = Field(min_length=1, max_length=8000)


class ChatRequest(BaseModel):
    messages: list[Message] = Field(min_length=1, max_length=30)


@app.get("/health")
async def health() -> dict:
    return {"status": "ok"}


@app.post("/chat")
async def chat(payload: ChatRequest, llm: AsyncLLMClient = Depends(get_llm)) -> dict:
    try:
        result = await llm.chat([message.model_dump() for message in payload.messages])
        return {
            "answer": result.text,
            "model": result.model,
            "usage": {"prompt_tokens": result.prompt_tokens, "completion_tokens": result.completion_tokens},
        }
    except httpx.TimeoutException as exc:
        raise HTTPException(504, "上游模型响应超时") from exc
    except httpx.HTTPStatusError as exc:
        raise HTTPException(502, "上游模型请求失败") from exc

Pydantic 校验失败由 FastAPI 自动返回 422。上游 HTTP 错误映射为 502，超时映射为 504，不把密钥、URL 和响应原文泄露给客户端。

04. 2.3 Day 08｜SSE 流式输出与多轮会话 ▼

2.3.1 SSE 最小协议

SSE 由服务端通过一个长连接持续发送事件。每个事件以空行结束：

event: token
data: {"text":"你"}

event: done
data: {}

JSON 编码可避免换行和中文造成协议边界混乱。

2.3.2 解析上游流

python
import json
from collections.abc import AsyncIterator


async def stream_chat(client: AsyncLLMClient, messages: list[dict[str, str]]) -> AsyncIterator[str]:
    payload = {"model": client.config.model, "messages": messages, "stream": True}
    async with client.client.stream("POST", "/chat/completions", json=payload) as response:
        response.raise_for_status()
        async for line in response.aiter_lines():
            if not line.startswith("data:"):
                continue
            data = line.removeprefix("data:").strip()
            if data == "[DONE]":
                break
            chunk = json.loads(data)
            text = chunk["choices"][0].get("delta", {}).get("content")
            if text:
                yield text

离开 async with 后 HTTPX 自动关闭上游响应，避免连接泄漏。

2.3.3 向浏览器转发 SSE

python
from fastapi.responses import StreamingResponse


def sse(event: str, data: dict) -> str:
    return f"event: {event}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"


@app.post("/chat/stream")
async def chat_stream(payload: ChatRequest, llm: AsyncLLMClient = Depends(get_llm)):
    messages = [message.model_dump() for message in payload.messages]

    async def events():
        try:
            async for token in stream_chat(llm, messages):
                yield sse("token", {"text": token})
            yield sse("done", {})
        except Exception:
            yield sse("error", {"message": "流式响应中断"})

    return StreamingResponse(events(), media_type="text/event-stream", headers={"Cache-Control": "no-cache"})

浏览器原生 EventSource 只直接支持 GET。本课接口使用 POST 传递 JSON，因此前端应使用 fetch 读取响应流，或把接口重新设计为 GET + 短期任务 ID；不能因为响应类型叫 SSE 就默认 EventSource 能直接调用。

javascript

生产前端还需缓存未完成的一行，并按空行拆分完整 SSE 事件。此处只用于观察“字节块逐次到达”。

2.3.4 会话存储与上下文裁剪

python
import asyncio
from collections import defaultdict


class SessionStore:
    def __init__(self, max_messages: int = 20):
        self.max_messages = max_messages
        self.data: dict[str, list[dict[str, str]]] = defaultdict(list)
        self.lock = asyncio.Lock()

    async def append_turn(self, session_id: str, user_text: str, assistant_text: str) -> None:
        async with self.lock:
            history = self.data[session_id]
            history.extend([
                {"role": "user", "content": user_text},
                {"role": "assistant", "content": assistant_text},
            ])
            self.data[session_id] = history[-self.max_messages:]

    async def get(self, session_id: str) -> list[dict[str, str]]:
        async with self.lock:
            return list(self.data.get(session_id, []))

这是单进程教学版。多 worker 部署需要 Redis 或数据库；按消息条数裁剪也只是入门方案，后续应改用 token 预算和摘要。

05. 2.4 Day 09｜工程稳定性与极简抽象 ▼

2.4.1 带退避的有限重试

python
import random


RETRYABLE_STATUS = {429, 500, 502, 503, 504}


async def call_with_retry(operation, attempts: int = 3):
    for attempt in range(1, attempts + 1):
        try:
            return await operation()
        except httpx.HTTPStatusError as exc:
            if exc.response.status_code not in RETRYABLE_STATUS or attempt == attempts:
                raise
        except (httpx.ConnectError, httpx.TimeoutException):
            if attempt == attempts:
                raise
        delay = min(4.0, 0.5 * 2 ** (attempt - 1)) + random.uniform(0, 0.2)
        await asyncio.sleep(delay)
    raise RuntimeError("不可达分支")

401、403 和大部分 400 错误不应重试；重试次数必须有限，并加入随机抖动，避免大量请求同时再次冲击上游。

2.4.2 进程内限流器

python
from collections import defaultdict, deque
from time import monotonic


class SlidingWindowLimiter:
    def __init__(self, limit: int, window_seconds: float):
        self.limit = limit
        self.window = window_seconds
        self.records: dict[str, deque[float]] = defaultdict(deque)
        self.lock = asyncio.Lock()

    async def allow(self, key: str) -> bool:
        now = monotonic()
        async with self.lock:
            queue = self.records[key]
            while queue and queue[0] <= now - self.window:
                queue.popleft()
            if len(queue) >= self.limit:
                return False
            queue.append(now)
            return True

多实例部署时进程内限流不共享状态，应迁移到网关或 Redis。key 可来自登录用户或可信网关传来的客户端标识，不能盲目信任任意请求头。

2.4.3 安全文件上传

python
from pathlib import Path
from uuid import uuid4
from fastapi import File, UploadFile


UPLOAD_DIR = Path("data/uploads").resolve()
UPLOAD_DIR.mkdir(parents=True, exist_ok=True)
ALLOWED_SUFFIXES = {".pdf", ".docx", ".txt", ".md"}
MAX_BYTES = 10 * 1024 * 1024


@app.post("/upload")
async def upload(file: UploadFile = File(...)) -> dict:
    suffix = Path(file.filename or "").suffix.lower()
    if suffix not in ALLOWED_SUFFIXES:
        raise HTTPException(415, "不支持的文件类型")
    content = await file.read(MAX_BYTES + 1)
    if len(content) > MAX_BYTES:
        raise HTTPException(413, "文件超过 10MB")
    safe_name = f"{uuid4().hex}{suffix}"
    target = UPLOAD_DIR / safe_name
    target.write_bytes(content)
    return {"file_id": safe_name, "size": len(content)}

不使用用户原始文件名作为保存路径。扩展名检查只是第一层，生产环境还要检查 MIME、文件签名、恶意内容并使用隔离存储。

async def observed_chat(llm: AsyncLLMClient, messages: list[dict[str, str]]) -> ChatResult:
    request_id = uuid4().hex
    started = perf_counter()
    try:
        result = await call_with_retry(lambda: llm.chat(messages))

2.4.5 极简框架抽象

python
from typing import Protocol


class ChatModel(Protocol):
    async def chat(self, messages: list[dict[str, str]]) -> ChatResult: ...


class OutputParser(Protocol):
    def parse(self, text: str): ...


async def run_chain(model: ChatModel, parser: OutputParser, messages: list[dict[str, str]]):
    result = await model.chat(messages)
    return parser.parse(result.text)

学生手写后应理解：框架主要提供标准接口、组合、回调与集成，不会替代需求分析、错误处理和评测。

2.5.3 20 并发压测脚本

python
import asyncio
from statistics import median
from time import perf_counter
import httpx


async def one_request(client: httpx.AsyncClient, index: int) -> dict:
    started = perf_counter()
    response = await client.post("/chat", json={"messages": [{"role": "user", "content": f"用一句话解释异步编程，编号 {index}"}]})
    return {"status": response.status_code, "latency": perf_counter() - started}


async def load_test(total: int = 20) -> None:
    async with httpx.AsyncClient(base_url="http://127.0.0.1:8000", timeout=90) as client:
        rows = await asyncio.gather(*(one_request(client, i) for i in range(total)))
    latencies = sorted(row["latency"] for row in rows)
    p95 = latencies[max(0, int(len(latencies) * 0.95) - 1)]
    print({
        "total": total,
        "success": sum(row["status"] == 200 for row in rows),
        "median_seconds": round(median(latencies), 3),
        "p95_seconds": round(p95, 3),
        "statuses": {code: sum(row["status"] == code for row in rows) for code in sorted({row["status"] for row in rows})},
    })

07. 2.6 常见错误与排查顺序 ▼
